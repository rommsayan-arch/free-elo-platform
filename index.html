function finishStreamMatch(winnerNum) { const p1Name = document.getElementById('streamMatchPlayer1').value; const p2Name = document.getElementById('streamMatchPlayer2').value; if (p1Name === p2Name) { alert("Игроки не могут совпадать!"); return; } const p1 = players.find(p => p.name === p1Name); const p2 = players.find(p => p.name === p2Name); if (!p1 || !p2) return; p1.matches++; p2.matches++; if (winnerNum === 1) { p1.wins++; p1.elo += 26; p2.elo = Math.max(0, p2.elo - 32); document.getElementById('streamMatchResult').innerHTML = `Победитель матча: ${p1Name} (+26 ELO), ${p2Name} (-32 ELO)`; } else { p2.wins++; p2.elo += 26; p1.elo = Math.max(0, p1.elo - 32); document.getElementById('streamMatchResult').innerHTML = `Победитель матча: ${p2Name} (+26 ELO), ${p1Name} (-32 ELO)`; } // Передаем данные для OBS режима в localStorage streamData.player1 = p1Name; streamData.player2 = p2Name; streamData.clan1 = p1.clan; streamData.clan2 = p2.clan; saveData(); updateUI(); }
